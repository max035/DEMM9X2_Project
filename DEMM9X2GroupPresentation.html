import numpy as np
import matplotlib.pyplot as plt

years = [2022, 2023, 2024, 2025, 2026]
x = np.arange(len(years))  # positions for groups
bar_width = 0.12           # width of each bar

# Equity components
share_capital = [6, 6, 6, 3883, 3883]
retained_earnings = [4717.3, 4685.2, 646, 7301, 6588]

# Debt components
borrowings = [6803.1,  5669.4, 11445, 1150, 200]
lease_liabilities = [16087.9, 17604, 19706, 19120, 18535]
trade_payables = [12976.4, 14661, 15354, 15036, 15326]

plt.figure(figsize=(14,7))

# Plot equity
plt.bar(x - 2*bar_width, share_capital, width=bar_width, label="Share Capital", color="#1f77b4")
plt.bar(x - bar_width, retained_earnings, width=bar_width, label="Retained Earnings", color="#ff7f0e")

plt.bar(x + 2*bar_width, trade_payables, width=bar_width, label="Trade & Payables", color="#9467bd")

plt.xlabel("Year")
plt.ylabel("Rm")
plt.title("Sources of Finance (Grouped Bar Chart, 2022–2026)")

import matplotlib.pyplot as plt

years = [2022, 2023, 2024, 2025, 2026]
debt_to_equity = [6.160612, 6.28535, -106.31741, 1.84861, 1.75000]  # Debt-to-Equity ratios
debt_ratio = [0.90790, 0.91260, 0.99375, 0.76574, 0.76469]
equity_ratio = [0.0921, 0.08740, 0.00625, 0.23426, 0.23531]

colors = ["#d62728", "#1f77b4"]  # Debt, Equity

for i in range(0, len(years), 2):

    fig, axes = plt.subplots(1, 2, figsize=(10, 5))

    # First chart
    values = [debt_ratio[i], equity_ratio[i]]
    axes[0].pie(values,
                labels=["Debt", "Equity"],
                autopct="%1.1f%%",
                startangle=90,
                colors=colors)
    axes[0].set_title(f"{years[i]}")

    # Second chart (if it exists)
    if i + 1 < len(years):
        values = [debt_ratio[i+1], equity_ratio[i+1]]
        axes[1].pie(values,
                    labels=["Debt", "Equity"],
                    autopct="%1.1f%%",
                    startangle=90,
                    colors=colors)
        axes[1].set_title(f"{years[i+1]}")
    else:
        axes[1].axis("off")

    plt.suptitle("Debt vs Equity")
    plt.tight_layout()
    plt.show()
